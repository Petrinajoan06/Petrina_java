/File read -- will give ASCII value only
package Joan;
import java.util.Scanner;
import java.io.*;
import java.util.*;

public class Main {
	public static void main(String[] args) {
		
			File f=new File("C:\\Users\\staff\\Desktop\\Files\\pet.txt");
			FileReader fr;
			try {
				fr = new FileReader(f);
				int a=fr.read();
				System.out.println((char)a);
				//will only read character by character
				while(a!=-1) {
					System.out.println(a);
					a=fr.read();//till this only ASCII value will come
				}
				
			}
			catch (Exception e) {
				System.out.println(e);
			}
	}
}

//square of the even digits and sum all digits
package Joan;
import java.util.Scanner;
import java.io.*;
import java.util.*;

public class Main {
	public static void main(String[] args) {
			
			int a=12345;
			int sum=0;
			while(a>0) {
				int t=a%10;
				if(a%2==0) {	
					sum+=t*t;
				}
				else {
					sum=sum+t;
				}
				a=a/10;
			}
			System.out.println(sum);
			
	}
}
