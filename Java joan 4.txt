//two dimensional array using functions
import java.util.Scanner;

class Main {
    void twodim(){
        int ar[][]=new int[3][3];
        Scanner s=new Scanner(System.in);
        for(int r=0;r<3;r++){
            for(int c=0;c<3;c++){
                System.out.println("enter array elements:");
                ar[r][c]=s.nextInt();
            }
        }
        System.out.println("after inserting:");
        for(int r=0;r<3;r++){
            for(int c=0;c<3;c++){
                System.out.print(ar[r][c]+" ");
            }
            System.out.println();
        }
    }
    public static void main(String[] args) {
        Main pet=new Main();
        pet.twodim();
    }
}

package Joan;

public class Main {
	void reversearray() {
		int[] arr= {3,5,7,2,8};
		
		
		for(int i=arr.length-1;i>=0;i--) {
			System.out.println(arr[i]);
		}
	}

		Main pet=new Main();
		pet.reversearray();

	}

//multiplication of two array
package Joan;

import java.util.Scanner;

public class Main {
	Scanner s=new Scanner(System.in);

	void multiplyTwoArray() {
		int arr1[][]=new int[3][3];
		int arr2[][]=new int[3][3];
		int arr3[][]=new int[3][3];
		
		//arr1 element storing
		
		for(int i=0;i<3;i++) {
			for(int j=0;j<3;j++) {
				System.out.println("enter elements");
				arr1[i][j]=s.nextInt();
			}
		}
		
		//arr2 element storing
		
		for(int i=0;i<3;i++) {
			for(int j=0;j<3;j++) {
				System.out.println("enter elements");
				arr2[i][j]=s.nextInt();
			}
		} 	
		
		//multiplication
		for(int i=0;i<3;i++) {
			for(int j=0;j<3;j++) {
				for(int k=0;k<3;k++) {
					arr3[i][j]+=arr1[i][k]*arr2[k][j];
				}
			}
		}
		
		//print element
		for(int i=0;i<3;i++) {
			for(int j=0;j<3;j++) 
			{
				System.out.print(arr3[i][j]+" ");
				//arr3[i][j]+=arr1[i][k]*arr2[k][j];
			}
			System.out.println();
		}
	}
	public static void main(String[] args) {
		Main pet=new Main();
		pet.multiplyTwoArray();

	}

}
