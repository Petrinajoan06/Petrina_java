//count vowels in a word
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner pet=new Scanner(System.in);
        System.out.println("enter string");
        String s1=pet.nextLine();
        int c=0;
        
        for(int i=0;i<s1.length();i++){
            if(s1.charAt(i)=='a'||s1.charAt(i)=='e'||s1.charAt(i)=='i'||s1.charAt(i)=='o'||s1.charAt(i)=='o'||s1.charAt(i)=='u')
            {
                c++;
            }
        }
        System.out.println(c);
    }
}

//factorial using functions
import java.util.Scanner;

class Main {
    int recur(int num)
    {
        if(num==1){
            return 1;
        }
        else{
            return num*recur(num-1);
        }
    }
    public static void main(String[] args) {
        Main pet=new Main();
        System.out.println(pet.recur(3));
    }
}

//prime using functions with return
import java.util.Scanner;

class Main {
    int prime(int n)
    {
        int h=0;
        for(int i=1;i<=n;i++)
        {
            if(n%i==0)
            {
                h++;
            }
        }
        return h;
    }
    public static void main(String[] args) {
        Main pet=new Main();
        Scanner sc=new Scanner(System.in);
        System.out.println("enter number");
        int i=sc.nextInt();
        
        int ans=pet.prime(i);
        if(ans==2){
            System.out.println("it is prime");
        }
        else{
            System.out.println("it is not prime");
        }
    }
}

//two dimensional array
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        int ar[][]=new int[3][3];
        Scanner s=new Scanner(System.in);
        for(int r=0;r<3;r++){
            for(int c=0;c<3;c++){
                System.out.println("enter array elements:");
                ar[r][c]=s.nextInt();
            }
        }
        System.out.println("after inserting:");
        for(int r=0;r<3;r++){
            for(int c=0;c<3;c++){
                System.out.print(ar[r][c]+" ");
            }
            System.out.println();
        }

    }
}

//three dimensional array
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        int ar[][][]=new int[2][2][2];
        Scanner s=new Scanner(System.in);
        for(int r=0;r<2;r++){
            for(int c=0;c<2;c++){
                for(int d=0;d<2;d++){
                    System.out.println("enter array elements:");
                    ar[r][c][d]=s.nextInt();
                }
            }
        }
        System.out.println("after inserting:");
        for(int r=0;r<2;r++){
            for(int c=0;c<2;c++){
                for(int d=0;d<2;d++){
                    System.out.print(ar[r][c][d]+" ");
                }
                System.out.println();
            }
            System.out.println();
        }
    }
}
